ايش هو الـKernel CallBacks؟

ببساطة هي دالة يستدعيها كيرنل الوندوز عند حدوث حدث معين مثل OpenProcess او إنشاء Thread او تشغيل برنامج او تشغيل درايفر او التعديل على الـRegistry وهذي الميزة تقدر تضيفها للدرايفر الخاص فيك وهذي الميزات تستخدم من قبل الـAnti Cheat و الـEDR's لتشغيل دالة معينة عند حدوث OpenProcess او غيرها لمراقبة العمليات المشبوهة ولحماية اللعبة.

انواع الـKernel CallBacks

PsSetCreateProcessNotifyRoutine تسجيل دالة مراقبة لإشعار الدرايفر عند تشغيل اي برنامج (Process) او إغلاقه

PsSetCreateThreadNotifyRoutine تسجيل دالة مراقبة لإشعار الدرايفر عند إنشاء Thread في اي (Process) او إغلاقه

PsSetLoadImageNotifyRoutine تسجيل دالة مراقبة لإشعار الدرايفر عند تشغيل درايفر آخر على النظام او DLL في اي Process

ObRegisterCallbacks تسجيل دالة مراقبة لإشعار الدرايفر اذا اي Process او Thread سوا OpenProcess على اي Process اخر وهذي تقريبا اهم ميزة لحماية اللعبة

الرسمة تحت توضح فكرة كل وحدة منهم.

static VOID PGCopyWide(
    _Out_writes_z_(Cch) WCHAR* Destination,
    _In_ size_t Cch,
    _In_opt_ PCWSTR Source)
{
    size_t i = 0;

    if (Destination == NULL || Cch == 0) {
        return;
    }

    if (Source != NULL) {
        while (i + 1 < Cch && Source[i] != L'\0') {
            Destination[i] = Source[i];
            i++;
        }
    }

    Destination[i] = L'\0';
}

/* Non-const PUNICODE_STRING to match PUNICODE_STRING call sites exactly */
static VOID PGCopyUnicode(
    _Out_writes_z_(Cch) WCHAR* Destination,
    _In_ size_t Cch,
    _In_opt_ PCUNICODE_STRING Source)
{
    size_t count = 0;

    if (Destination == NULL || Cch == 0) {
        return;
    }

    Destination[0] = L'\0';

    if (Source == NULL || Source->Buffer == NULL || Source->Length == 0) {
        return;
    }

    count = Source->Length / sizeof(WCHAR);
    if (count >= Cch) {
        count = Cch - 1;
    }

    RtlCopyMemory(Destination, Source->Buffer, count * sizeof(WCHAR));
    Destination[count] = L'\0';
}

    Destination[i] = L'\0';
}

    _In_opt_ PEPROCESS Process,
    _Out_writes_z_(Cch) WCHAR* Destination,
    _In_ size_t Cch)

    if (Process == NULL) {

static VOID PGInitAlert(_Out_ PG_ALERT* Alert, _In_ ULONG Type)
{
    LARGE_INTEGER time;

    RtlZeroMemory(Alert, sizeof(*Alert));
    KeQuerySystemTime(&time);

    Alert->Version = 1;
    Alert->Type = Type;
    Alert->Timestamp = (ULONGLONG)time.QuadPart;
}

static VOID PGPushAlert(_In_ PG_ALERT* Alert)
{
    KIRQL irql;
    ULONG index;

    if (g_AlertRing == NULL) {
        return;
    }

    KeAcquireSpinLock(&g_AlertRing->Lock, &irql);

    if (g_AlertRing->Count == PG_ALERT_MAX) {
        g_AlertRing->Tail = (g_AlertRing->Tail + 1) % PG_ALERT_MAX;
        g_AlertRing->Count--;
    }

    index = g_AlertRing->Head;
    g_AlertRing->Entries[index] = *Alert;
    g_AlertRing->Head = (index + 1) % PG_ALERT_MAX;
    g_AlertRing->Count++;

    KeReleaseSpinLock(&g_AlertRing->Lock, irql);
}

static VOID PGProcessNotify(
    PEPROCESS Process,
    HANDLE ProcessId,
    PPS_CREATE_NOTIFY_INFO CreateInfo)
{
    PG_ALERT alert;

    PGInitAlert(&alert, CreateInfo != NULL ? PG_ALERT_PROCESS_CREATE : PG_ALERT_PROCESS_EXIT);
    alert.ProcessId = (ULONG)(ULONG_PTR)ProcessId;
    alert.Create = (BOOLEAN)(CreateInfo != NULL);

    if (CreateInfo != NULL) {
        PGCopyUnicode(alert.ImageName, RTL_NUMBER_OF(alert.ImageName), CreateInfo->ImageFileName);
    }

    PGGetProcessName(Process, alert.ProcessName, RTL_NUMBER_OF(alert.ProcessName));
    if (alert.ImageName[0] == L'\0') {
        PGCopyWide(alert.ImageName, RTL_NUMBER_OF(alert.ImageName), alert.ProcessName);
    }

    PGCopyWide(
        alert.Message,
        RTL_NUMBER_OF(alert.Message),
        CreateInfo != NULL ? L"Process created" : L"Process exited");

    PGPushAlert(&alert);
}

static VOID PGImageNotify(
    PUNICODE_STRING FullImageName,
    HANDLE ProcessId,
    PIMAGE_INFO ImageInfo)
{
    PG_ALERT alert;

    PGInitAlert(&alert, PG_ALERT_IMAGE_LOAD);
    alert.ProcessId = (ULONG)(ULONG_PTR)ProcessId;

    PGCopyUnicode(alert.ImageName, RTL_NUMBER_OF(alert.ImageName), FullImageName);

    if (ImageInfo != NULL) {
        alert.ImageBase = (ULONGLONG)(ULONG_PTR)ImageInfo->ImageBase;
        alert.ImageSize = (ULONG)ImageInfo->ImageSize;
        alert.SystemModeImage = (BOOLEAN)(ImageInfo->SystemModeImage != 0);
    }

    if (ProcessId != NULL) {
        PEPROCESS process = NULL;

        if (NT_SUCCESS(PsLookupProcessByProcessId(ProcessId, &process))) {
            PGGetProcessName(process, alert.ProcessName, RTL_NUMBER_OF(alert.ProcessName));
            ObDereferenceObject(process);
        }
    }

    PGCopyWide(
        alert.Message,
        RTL_NUMBER_OF(alert.Message),
        alert.SystemModeImage ? L"System image loaded" : L"Image loaded");

    PGPushAlert(&alert);
}

static BOOLEAN PGGetAccessPointers(
    _In_ POB_PRE_OPERATION_INFORMATION Information,
    _Out_ PACCESS_MASK* DesiredAccess,
    _Out_ PACCESS_MASK* OriginalAccess)
{
    *DesiredAccess = NULL;
    *OriginalAccess = NULL;

    if (Information->Operation == OB_OPERATION_HANDLE_CREATE) {
        *DesiredAccess = &Information->Parameters->CreateHandleInformation.DesiredAccess;
        *OriginalAccess = &Information->Parameters->CreateHandleInformation.OriginalDesiredAccess;
        return TRUE;
    }

    if (Information->Operation == OB_OPERATION_HANDLE_DUPLICATE) {
        *DesiredAccess = &Information->Parameters->DuplicateHandleInformation.DesiredAccess;
        *OriginalAccess = &Information->Parameters->DuplicateHandleInformation.OriginalDesiredAccess;
        return TRUE;
    }

    return FALSE;
}

static NTSTATUS PGDeviceControl(
    _In_ PDEVICE_OBJECT DeviceObject,
    _In_ PIRP Irp)
{
    PIO_STACK_LOCATION stack;
    NTSTATUS status = STATUS_INVALID_DEVICE_REQUEST;
    ULONG information = 0;
    ULONG code;
    ULONG inputLength;
    ULONG outputLength;
    PVOID buffer;

    UNREFERENCED_PARAMETER(DeviceObject);

    stack = IoGetCurrentIrpStackLocation(Irp);
    code = stack->Parameters.DeviceIoControl.IoControlCode;
    inputLength = stack->Parameters.DeviceIoControl.InputBufferLength;
    outputLength = stack->Parameters.DeviceIoControl.OutputBufferLength;
    buffer = Irp->AssociatedIrp.SystemBuffer;

    switch (code) {
    case IOCTL_PG_SET_PID:
        if (inputLength < sizeof(ULONG)) {
            status = STATUS_BUFFER_TOO_SMALL;
        }
        else {
            ULONG pid = *(ULONG*)buffer;
            PG_ALERT alert;

            InterlockedExchange((volatile LONG*)&g_ProtectedPid, (LONG)pid);

            PGInitAlert(&alert, PG_ALERT_CONTROL);
            alert.ProcessId = (ULONG)(ULONG_PTR)PsGetCurrentProcessId();
            alert.TargetProcessId = pid;
            alert.Create = (BOOLEAN)(pid != 0);
            PGCopyWide(alert.Message, RTL_NUMBER_OF(alert.Message), L"Protected PID updated");
            PGPushAlert(&alert);

            status = STATUS_SUCCESS;
        }
        break;

    case IOCTL_PG_GET_ALERTS:
        if (outputLength < sizeof(PG_ALERT)) {
            status = STATUS_BUFFER_TOO_SMALL;
        }
        else if (g_AlertRing == NULL) {
            status = STATUS_DEVICE_NOT_READY;
        }
        else {
            ULONG maxAlerts = outputLength / sizeof(PG_ALERT);
            ULONG count;
            KIRQL irql;
            ULONG i;

            if (maxAlerts > PG_ALERT_MAX) {
                maxAlerts = PG_ALERT_MAX;
            }

            KeAcquireSpinLock(&g_AlertRing->Lock, &irql);
            count = g_AlertRing->Count;
            if (count > maxAlerts) {
                count = maxAlerts;
            }

            for (i = 0; i < count; i++) {
                ((PG_ALERT*)buffer)[i] = g_AlertRing->Entries[g_AlertRing->Tail];
                g_AlertRing->Tail = (g_AlertRing->Tail + 1) % PG_ALERT_MAX;
                g_AlertRing->Count--;
            }
            KeReleaseSpinLock(&g_AlertRing->Lock, irql);

            information = count * sizeof(PG_ALERT);
            status = STATUS_SUCCESS;
        }
        break;

    case IOCTL_PG_CLEAR_ALERTS:
        if (g_AlertRing != NULL) {
            KIRQL irql;

            KeAcquireSpinLock(&g_AlertRing->Lock, &irql);
            g_AlertRing->Head = 0;
            g_AlertRing->Tail = 0;
            g_AlertRing->Count = 0;
            KeReleaseSpinLock(&g_AlertRing->Lock, irql);
        }
        status = STATUS_SUCCESS;
        break;

    case IOCTL_PG_GET_STATUS:
        if (outputLength < sizeof(PG_STATUS)) {
            status = STATUS_BUFFER_TOO_SMALL;
        }
        else if (g_AlertRing == NULL) {
            status = STATUS_DEVICE_NOT_READY;
        }
        else {
            PG_STATUS* pgStatus = (PG_STATUS*)buffer;
            KIRQL irql;

            RtlZeroMemory(pgStatus, sizeof(*pgStatus));
            pgStatus->ProtectedPid = (ULONG)InterlockedCompareExchange((volatile LONG*)&g_ProtectedPid, 0, 0);
            pgStatus->MaxAlerts = PG_ALERT_MAX;

            KeAcquireSpinLock(&g_AlertRing->Lock, &irql);
            pgStatus->AlertCount = g_AlertRing->Count;
            KeReleaseSpinLock(&g_AlertRing->Lock, irql);

            information = sizeof(PG_STATUS);
            status = STATUS_SUCCESS;
        }
        break;

    default:
        status = STATUS_INVALID_DEVICE_REQUEST;
        break;
    }

    Irp->IoStatus.Status = status;
    Irp->IoStatus.Information = information;
    IoCompleteRequest(Irp, IO_NO_INCREMENT);
    return status;
}

static VOID PGUnload(_In_ PDRIVER_OBJECT DriverObject)
{
    UNICODE_STRING symlinkName;

    UNREFERENCED_PARAMETER(DriverObject);

    if (g_ObRegistered && g_ObRegistration != NULL) {
        ObUnRegisterCallbacks(g_ObRegistration);
        g_ObRegistration = NULL;
        g_ObRegistered = FALSE;
    }

    if (g_ProcessNotifyRegistered) {
        PsSetCreateProcessNotifyRoutineEx(PGProcessNotify, TRUE);
        g_ProcessNotifyRegistered = FALSE;
    }

    if (g_ImageNotifyRegistered) {
        PsRemoveLoadImageNotifyRoutine(PGImageNotify);
        g_ImageNotifyRegistered = FALSE;
    }

    if (g_SymlinkCreated) {
        RtlInitUnicodeString(&symlinkName, PG_SYMLINK_NAME);
        IoDeleteSymbolicLink(&symlinkName);
        g_SymlinkCreated = FALSE;
    }

    if (g_DeviceObject != NULL) {
        IoDeleteDevice(g_DeviceObject);
        g_DeviceObject = NULL;
    }

    if (g_AlertRing != NULL) {
        ExFreePoolWithTag(g_AlertRing, PG_TAG);
        g_AlertRing = NULL;
    }
}

NTSTATUS DriverEntry(
    _In_ PDRIVER_OBJECT DriverObject,
    _In_ PUNICODE_STRING RegistryPath)
{
    UNICODE_STRING deviceName;
    UNICODE_STRING symlinkName;
    UNICODE_STRING altitude;
    OB_OPERATION_REGISTRATION operations[2];
    OB_CALLBACK_REGISTRATION registration;
    NTSTATUS status;

    UNREFERENCED_PARAMETER(RegistryPath);

    RtlInitUnicodeString(&deviceName, PG_DEVICE_NAME);
    RtlInitUnicodeString(&symlinkName, PG_SYMLINK_NAME);
    RtlInitUnicodeString(&altitude, PG_ALTITUDE);

    g_AlertRing = (PG_ALERT_RING*)ExAllocatePool2(
        POOL_FLAG_NON_PAGED, sizeof(PG_ALERT_RING), PG_TAG);

    if (g_AlertRing == NULL) {
        return STATUS_INSUFFICIENT_RESOURCES;
    }

    KeInitializeSpinLock(&g_AlertRing->Lock);

    status = IoCreateDevice(
        DriverObject,
        0,
        &deviceName,
        FILE_DEVICE_UNKNOWN,
        FILE_DEVICE_SECURE_OPEN,
        FALSE,
        &g_DeviceObject);

    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }

    status = IoCreateSymbolicLink(&symlinkName, &deviceName);
    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }
    g_SymlinkCreated = TRUE;

    DriverObject->MajorFunction[IRP_MJ_CREATE] = PGCreateClose;
    DriverObject->MajorFunction[IRP_MJ_CLOSE] = PGCreateClose;
    DriverObject->MajorFunction[IRP_MJ_DEVICE_CONTROL] = PGDeviceControl;
    DriverObject->DriverUnload = PGUnload;

    status = PsSetCreateProcessNotifyRoutineEx(PGProcessNotify, FALSE);
    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }
    g_ProcessNotifyRegistered = TRUE;

    status = PsSetLoadImageNotifyRoutine(PGImageNotify);
    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }
    g_ImageNotifyRegistered = TRUE;

    RtlZeroMemory(operations, sizeof(operations));
    operations[0].ObjectType = PsProcessType;
    operations[0].Operations = OB_OPERATION_HANDLE_CREATE | OB_OPERATION_HANDLE_DUPLICATE;
    operations[0].PreOperation = PGPreOperation;
    operations[0].PostOperation = PGPostOperation;

    operations[1].ObjectType = PsThreadType;
    operations[1].Operations = OB_OPERATION_HANDLE_CREATE | OB_OPERATION_HANDLE_DUPLICATE;
    operations[1].PreOperation = PGPreOperation;
    operations[1].PostOperation = PGPostOperation;

    RtlZeroMemory(&registration, sizeof(registration));
    registration.Version = OB_FLT_REGISTRATION_VERSION;
    registration.OperationRegistrationCount = 2;
    registration.Altitude = altitude;
    registration.OperationRegistration = operations;

    status = ObRegisterCallbacks(&registration, &g_ObRegistration);
    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }
    g_ObRegistered = TRUE;

    return STATUS_SUCCESS;
}

static VOID PGCopyWide(
    _Out_writes_z_(Cch) WCHAR* Destination,
    _In_ size_t Cch,
    _In_opt_ PCWSTR Source)
{
    size_t i = 0;

    if (Destination == NULL || Cch == 0) {
        return;
    }

    if (Source != NULL) {
        while (i + 1 < Cch && Source[i] != L'\0') {
            Destination[i] = Source[i];
            i++;
        }
    }

    Destination[i] = L'\0';
}

/* Non-const PUNICODE_STRING to match PUNICODE_STRING call sites exactly */
static VOID PGCopyUnicode(
    _Out_writes_z_(Cch) WCHAR* Destination,
    _In_ size_t Cch,
    _In_opt_ PCUNICODE_STRING Source)
{
    size_t count = 0;

    if (Destination == NULL || Cch == 0) {
        return;
    }

    Destination[0] = L'\0';

    if (Source == NULL || Source->Buffer == NULL || Source->Length == 0) {
        return;
    }

    count = Source->Length / sizeof(WCHAR);
    if (count >= Cch) {
        count = Cch - 1;
    }

    RtlCopyMemory(Destination, Source->Buffer, count * sizeof(WCHAR));
    Destination[count] = L'\0';
}

static VOID PGCopyAnsi(
    _Out_writes_z_(Cch) WCHAR* Destination,
    _In_ size_t Cch,
    _In_opt_ PCHAR Source)
{
    size_t i = 0;

    if (Destination == NULL || Cch == 0) {
        return;
    }

    if (Source != NULL) {
        while (i + 1 < Cch && Source[i] != '\0') {
            Destination[i] = (WCHAR)(UCHAR)Source[i];
            i++;
        }
    }

    Destination[i] = L'\0';
}

static VOID PGGetProcessName(
    _In_opt_ PEPROCESS Process,
    _Out_writes_z_(Cch) WCHAR* Destination,
    _In_ size_t Cch)
{
    if (Process == NULL) {
        PGCopyWide(Destination, Cch, L"<unknown>");
        return;
    }

    PGCopyAnsi(Destination, Cch, PsGetProcessImageFileName(Process));
}

static VOID PGInitAlert(_Out_ PG_ALERT* Alert, _In_ ULONG Type)
{
    LARGE_INTEGER time;

    RtlZeroMemory(Alert, sizeof(*Alert));
    KeQuerySystemTime(&time);

    Alert->Version = 1;
    Alert->Type = Type;
    Alert->Timestamp = (ULONGLONG)time.QuadPart;
}

static VOID PGPushAlert(_In_ PG_ALERT* Alert)
{
    KIRQL irql;
    ULONG index;

    if (g_AlertRing == NULL) {
        return;
    }

    KeAcquireSpinLock(&g_AlertRing->Lock, &irql);

    if (g_AlertRing->Count == PG_ALERT_MAX) {
        g_AlertRing->Tail = (g_AlertRing->Tail + 1) % PG_ALERT_MAX;
        g_AlertRing->Count--;
    }

    index = g_AlertRing->Head;
    g_AlertRing->Entries[index] = *Alert;
    g_AlertRing->Head = (index + 1) % PG_ALERT_MAX;
    g_AlertRing->Count++;

    KeReleaseSpinLock(&g_AlertRing->Lock, irql);
}

static VOID PGProcessNotify(
    PEPROCESS Process,
    HANDLE ProcessId,
    PPS_CREATE_NOTIFY_INFO CreateInfo)
{
    PG_ALERT alert;

    PGInitAlert(&alert, CreateInfo != NULL ? PG_ALERT_PROCESS_CREATE : PG_ALERT_PROCESS_EXIT);
    alert.ProcessId = (ULONG)(ULONG_PTR)ProcessId;
    alert.Create = (BOOLEAN)(CreateInfo != NULL);

    if (CreateInfo != NULL) {
        PGCopyUnicode(alert.ImageName, RTL_NUMBER_OF(alert.ImageName), CreateInfo->ImageFileName);
    }

    PGGetProcessName(Process, alert.ProcessName, RTL_NUMBER_OF(alert.ProcessName));
    if (alert.ImageName[0] == L'\0') {
        PGCopyWide(alert.ImageName, RTL_NUMBER_OF(alert.ImageName), alert.ProcessName);
    }

    PGCopyWide(
        alert.Message,
        RTL_NUMBER_OF(alert.Message),
        CreateInfo != NULL ? L"Process created" : L"Process exited");

    PGPushAlert(&alert);
}

static VOID PGImageNotify(
    PUNICODE_STRING FullImageName,
    HANDLE ProcessId,
    PIMAGE_INFO ImageInfo)
{
    PG_ALERT alert;

    PGInitAlert(&alert, PG_ALERT_IMAGE_LOAD);
    alert.ProcessId = (ULONG)(ULONG_PTR)ProcessId;

    PGCopyUnicode(alert.ImageName, RTL_NUMBER_OF(alert.ImageName), FullImageName);

    if (ImageInfo != NULL) {
        alert.ImageBase = (ULONGLONG)(ULONG_PTR)ImageInfo->ImageBase;
        alert.ImageSize = (ULONG)ImageInfo->ImageSize;
        alert.SystemModeImage = (BOOLEAN)(ImageInfo->SystemModeImage != 0);
    }

    if (ProcessId != NULL) {
        PEPROCESS process = NULL;

        if (NT_SUCCESS(PsLookupProcessByProcessId(ProcessId, &process))) {
            PGGetProcessName(process, alert.ProcessName, RTL_NUMBER_OF(alert.ProcessName));
            ObDereferenceObject(process);
        }
    }

    PGCopyWide(
        alert.Message,
        RTL_NUMBER_OF(alert.Message),
        alert.SystemModeImage ? L"System image loaded" : L"Image loaded");

    PGPushAlert(&alert);
}

static BOOLEAN PGGetAccessPointers(
    _In_ POB_PRE_OPERATION_INFORMATION Information,
    _Out_ PACCESS_MASK* DesiredAccess,
    _Out_ PACCESS_MASK* OriginalAccess)
{
    *DesiredAccess = NULL;
    *OriginalAccess = NULL;

    if (Information->Operation == OB_OPERATION_HANDLE_CREATE) {
        *DesiredAccess = &Information->Parameters->CreateHandleInformation.DesiredAccess;
        *OriginalAccess = &Information->Parameters->CreateHandleInformation.OriginalDesiredAccess;
        return TRUE;
    }

    if (Information->Operation == OB_OPERATION_HANDLE_DUPLICATE) {
        *DesiredAccess = &Information->Parameters->DuplicateHandleInformation.DesiredAccess;
        *OriginalAccess = &Information->Parameters->DuplicateHandleInformation.OriginalDesiredAccess;
        return TRUE;
    }

    return FALSE;
}

static VOID PGMakeHandleAlert(
    _In_ PG_ALERT* Alert,
    _In_ ULONG Type,
    _In_ ULONG TargetProcessId,
    _In_opt_ PEPROCESS TargetProcess,
    _In_ ACCESS_MASK OriginalAccess,
    _In_ ACCESS_MASK DesiredAccess,
    _In_ PCWSTR Message)
{
    PGInitAlert(Alert, Type);
    Alert->ProcessId = (ULONG)(ULONG_PTR)PsGetCurrentProcessId();
    Alert->TargetProcessId = TargetProcessId;
    Alert->OriginalAccess = (ULONG)OriginalAccess;
    Alert->DesiredAccess = (ULONG)DesiredAccess;

    PGGetProcessName(PsGetCurrentProcess(), Alert->ProcessName, RTL_NUMBER_OF(Alert->ProcessName));
    PGGetProcessName(TargetProcess, Alert->TargetName, RTL_NUMBER_OF(Alert->TargetName));
    PGCopyWide(Alert->Message, RTL_NUMBER_OF(Alert->Message), Message);
}

static OB_PREOP_CALLBACK_STATUS PGPreOperation(
    PVOID RegistrationContext,
    POB_PRE_OPERATION_INFORMATION Information)
{
    ULONG protectedPid;

    UNREFERENCED_PARAMETER(RegistrationContext);

    if (Information->KernelHandle) {
        return OB_PREOP_SUCCESS;
    }

    protectedPid = (ULONG)InterlockedCompareExchange((volatile LONG*)&g_ProtectedPid, 0, 0);
    if (protectedPid == 0) {
        return OB_PREOP_SUCCESS;
    }

    if (Information->ObjectType == *PsProcessType) {
        PEPROCESS target = (PEPROCESS)Information->Object;
        ULONG targetPid = (ULONG)(ULONG_PTR)PsGetProcessId(target);
        PACCESS_MASK desired = NULL;
        PACCESS_MASK original = NULL;
        PG_ALERT alert;

        if (targetPid != protectedPid) {
            return OB_PREOP_SUCCESS;
        }

        if ((ULONG)(ULONG_PTR)PsGetCurrentProcessId() == protectedPid) {
            return OB_PREOP_SUCCESS;
        }

        if (!PGGetAccessPointers(Information, &desired, &original)) {
            return OB_PREOP_SUCCESS;
        }

        PGMakeHandleAlert(
            &alert,
            PG_ALERT_HANDLE_PRE,
            targetPid,
            target,
            *original,
            *desired & ~PG_DENY_PROCESS_ACCESS,
            L"Process handle access stripped");

        *desired &= ~PG_DENY_PROCESS_ACCESS;
        alert.DesiredAccess = (ULONG)*desired;
        PGPushAlert(&alert);
    }
    else if (Information->ObjectType == *PsThreadType) {
        PETHREAD thread = (PETHREAD)Information->Object;
        PEPROCESS target = PsGetThreadProcess(thread);
        ULONG targetPid = target != NULL ? (ULONG)(ULONG_PTR)PsGetProcessId(target) : 0;
        PACCESS_MASK desired = NULL;
        PACCESS_MASK original = NULL;
        PG_ALERT alert;

        if (targetPid != protectedPid) {
            return OB_PREOP_SUCCESS;
        }

        if ((ULONG)(ULONG_PTR)PsGetCurrentProcessId() == protectedPid) {
            return OB_PREOP_SUCCESS;
        }

        if (!PGGetAccessPointers(Information, &desired, &original)) {
            return OB_PREOP_SUCCESS;
        }

        PGMakeHandleAlert(
            &alert,
            PG_ALERT_HANDLE_PRE,
            targetPid,
            target,
            *original,
            *desired & ~PG_DENY_THREAD_ACCESS,
            L"Thread handle access stripped");

        *desired &= ~PG_DENY_THREAD_ACCESS;
        alert.DesiredAccess = (ULONG)*desired;
        PGPushAlert(&alert);
    }

    return OB_PREOP_SUCCESS;
}

static VOID PGPostOperation(
    PVOID RegistrationContext,
    POB_POST_OPERATION_INFORMATION Information)
{
    ULONG protectedPid;

    UNREFERENCED_PARAMETER(RegistrationContext);

    if (Information == NULL || Information->Object == NULL) {
        return;
    }

    if (Information->KernelHandle) {
        return;
    }

    protectedPid = (ULONG)InterlockedCompareExchange((volatile LONG*)&g_ProtectedPid, 0, 0);
    if (protectedPid == 0) {
        return;
    }

    if (Information->ObjectType == *PsProcessType) {
        PEPROCESS target = (PEPROCESS)Information->Object;
        ULONG targetPid = (ULONG)(ULONG_PTR)PsGetProcessId(target);
        PG_ALERT alert;

        if (targetPid != protectedPid) {
            return;
        }

        PGMakeHandleAlert(
            &alert,
            PG_ALERT_HANDLE_POST,
            targetPid,
            target,
            0,
            0,
            L"Process handle operation completed");
        alert.Status = Information->ReturnStatus;
        alert.Create = (BOOLEAN)(NT_SUCCESS(Information->ReturnStatus) ? TRUE : FALSE);
        PGPushAlert(&alert);
    }
    else if (Information->ObjectType == *PsThreadType) {
        PETHREAD thread = (PETHREAD)Information->Object;
        PEPROCESS target = PsGetThreadProcess(thread);
        ULONG targetPid = target != NULL ? (ULONG)(ULONG_PTR)PsGetProcessId(target) : 0;
        PG_ALERT alert;

        if (targetPid != protectedPid) {
            return;
        }

        PGMakeHandleAlert(
            &alert,
            PG_ALERT_HANDLE_POST,
            targetPid,
            target,
            0,
            0,
            L"Thread handle operation completed");
        alert.Status = Information->ReturnStatus;
        alert.Create = (BOOLEAN)(NT_SUCCESS(Information->ReturnStatus) ? TRUE : FALSE);
        PGPushAlert(&alert);
    }
}

static NTSTATUS PGDeviceControl(
    _In_ PDEVICE_OBJECT DeviceObject,
    _In_ PIRP Irp)
{
    PIO_STACK_LOCATION stack;
    NTSTATUS status = STATUS_INVALID_DEVICE_REQUEST;
    ULONG information = 0;
    ULONG code;
    ULONG inputLength;
    ULONG outputLength;
    PVOID buffer;

    UNREFERENCED_PARAMETER(DeviceObject);

    stack = IoGetCurrentIrpStackLocation(Irp);
    code = stack->Parameters.DeviceIoControl.IoControlCode;
    inputLength = stack->Parameters.DeviceIoControl.InputBufferLength;
    outputLength = stack->Parameters.DeviceIoControl.OutputBufferLength;
    buffer = Irp->AssociatedIrp.SystemBuffer;

    switch (code) {
    case IOCTL_PG_SET_PID:
        if (inputLength < sizeof(ULONG)) {
            status = STATUS_BUFFER_TOO_SMALL;
        }
        else {
            ULONG pid = *(ULONG*)buffer;
            PG_ALERT alert;

            InterlockedExchange((volatile LONG*)&g_ProtectedPid, (LONG)pid);

            PGInitAlert(&alert, PG_ALERT_CONTROL);
            alert.ProcessId = (ULONG)(ULONG_PTR)PsGetCurrentProcessId();
            alert.TargetProcessId = pid;
            alert.Create = (BOOLEAN)(pid != 0);
            PGCopyWide(alert.Message, RTL_NUMBER_OF(alert.Message), L"Protected PID updated");
            PGPushAlert(&alert);

            status = STATUS_SUCCESS;
        }
        break;

    case IOCTL_PG_GET_ALERTS:
        if (outputLength < sizeof(PG_ALERT)) {
            status = STATUS_BUFFER_TOO_SMALL;
        }
        else if (g_AlertRing == NULL) {
            status = STATUS_DEVICE_NOT_READY;
        }
        else {
            ULONG maxAlerts = outputLength / sizeof(PG_ALERT);
            ULONG count;
            KIRQL irql;
            ULONG i;

            if (maxAlerts > PG_ALERT_MAX) {
                maxAlerts = PG_ALERT_MAX;
            }

            KeAcquireSpinLock(&g_AlertRing->Lock, &irql);
            count = g_AlertRing->Count;
            if (count > maxAlerts) {
                count = maxAlerts;
            }

            for (i = 0; i < count; i++) {
                ((PG_ALERT*)buffer)[i] = g_AlertRing->Entries[g_AlertRing->Tail];
                g_AlertRing->Tail = (g_AlertRing->Tail + 1) % PG_ALERT_MAX;
                g_AlertRing->Count--;
            }
            KeReleaseSpinLock(&g_AlertRing->Lock, irql);

            information = count * sizeof(PG_ALERT);
            status = STATUS_SUCCESS;
        }
        break;

    case IOCTL_PG_CLEAR_ALERTS:
        if (g_AlertRing != NULL) {
            KIRQL irql;

            KeAcquireSpinLock(&g_AlertRing->Lock, &irql);
            g_AlertRing->Head = 0;
            g_AlertRing->Tail = 0;
            g_AlertRing->Count = 0;
            KeReleaseSpinLock(&g_AlertRing->Lock, irql);
        }
        status = STATUS_SUCCESS;
        break;

    case IOCTL_PG_GET_STATUS:
        if (outputLength < sizeof(PG_STATUS)) {
            status = STATUS_BUFFER_TOO_SMALL;
        }
        else if (g_AlertRing == NULL) {
            status = STATUS_DEVICE_NOT_READY;
        }
        else {
            PG_STATUS* pgStatus = (PG_STATUS*)buffer;
            KIRQL irql;

            RtlZeroMemory(pgStatus, sizeof(*pgStatus));
            pgStatus->ProtectedPid = (ULONG)InterlockedCompareExchange((volatile LONG*)&g_ProtectedPid, 0, 0);
            pgStatus->MaxAlerts = PG_ALERT_MAX;

            KeAcquireSpinLock(&g_AlertRing->Lock, &irql);
            pgStatus->AlertCount = g_AlertRing->Count;
            KeReleaseSpinLock(&g_AlertRing->Lock, irql);

            information = sizeof(PG_STATUS);
            status = STATUS_SUCCESS;
        }
        break;

    default:
        status = STATUS_INVALID_DEVICE_REQUEST;
        break;
    }

    Irp->IoStatus.Status = status;
    Irp->IoStatus.Information = information;
    IoCompleteRequest(Irp, IO_NO_INCREMENT);
    return status;
}

static NTSTATUS PGCreateClose(
    _In_ PDEVICE_OBJECT DeviceObject,
    _In_ PIRP Irp)
{
    UNREFERENCED_PARAMETER(DeviceObject);

    Irp->IoStatus.Status = STATUS_SUCCESS;
    Irp->IoStatus.Information = 0;
    IoCompleteRequest(Irp, IO_NO_INCREMENT);
    return STATUS_SUCCESS;
}

static VOID PGUnload(_In_ PDRIVER_OBJECT DriverObject)
{
    UNICODE_STRING symlinkName;

    UNREFERENCED_PARAMETER(DriverObject);

    if (g_ObRegistered && g_ObRegistration != NULL) {
        ObUnRegisterCallbacks(g_ObRegistration);
        g_ObRegistration = NULL;
        g_ObRegistered = FALSE;
    }

    if (g_ProcessNotifyRegistered) {
        PsSetCreateProcessNotifyRoutineEx(PGProcessNotify, TRUE);
        g_ProcessNotifyRegistered = FALSE;
    }

    if (g_ImageNotifyRegistered) {
        PsRemoveLoadImageNotifyRoutine(PGImageNotify);
        g_ImageNotifyRegistered = FALSE;
    }

    if (g_SymlinkCreated) {
        RtlInitUnicodeString(&symlinkName, PG_SYMLINK_NAME);
        IoDeleteSymbolicLink(&symlinkName);
        g_SymlinkCreated = FALSE;
    }

    if (g_DeviceObject != NULL) {
        IoDeleteDevice(g_DeviceObject);
        g_DeviceObject = NULL;
    }

    if (g_AlertRing != NULL) {
        ExFreePoolWithTag(g_AlertRing, PG_TAG);
        g_AlertRing = NULL;
    }
}

NTSTATUS DriverEntry(
    _In_ PDRIVER_OBJECT DriverObject,
    _In_ PUNICODE_STRING RegistryPath)
{
    UNICODE_STRING deviceName;
    UNICODE_STRING symlinkName;
    UNICODE_STRING altitude;
    OB_OPERATION_REGISTRATION operations[2];
    OB_CALLBACK_REGISTRATION registration;
    NTSTATUS status;

    UNREFERENCED_PARAMETER(RegistryPath);

    RtlInitUnicodeString(&deviceName, PG_DEVICE_NAME);
    RtlInitUnicodeString(&symlinkName, PG_SYMLINK_NAME);
    RtlInitUnicodeString(&altitude, PG_ALTITUDE);

    g_AlertRing = (PG_ALERT_RING*)ExAllocatePool2(
        POOL_FLAG_NON_PAGED, sizeof(PG_ALERT_RING), PG_TAG);

    if (g_AlertRing == NULL) {
        return STATUS_INSUFFICIENT_RESOURCES;
    }

    KeInitializeSpinLock(&g_AlertRing->Lock);

    status = IoCreateDevice(
        DriverObject,
        0,
        &deviceName,
        FILE_DEVICE_UNKNOWN,
        FILE_DEVICE_SECURE_OPEN,
        FALSE,
        &g_DeviceObject);

    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }

    status = IoCreateSymbolicLink(&symlinkName, &deviceName);
    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }
    g_SymlinkCreated = TRUE;

    DriverObject->MajorFunction[IRP_MJ_CREATE] = PGCreateClose;
    DriverObject->MajorFunction[IRP_MJ_CLOSE] = PGCreateClose;
    DriverObject->MajorFunction[IRP_MJ_DEVICE_CONTROL] = PGDeviceControl;
    DriverObject->DriverUnload = PGUnload;

    status = PsSetCreateProcessNotifyRoutineEx(PGProcessNotify, FALSE);
    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }
    g_ProcessNotifyRegistered = TRUE;

    status = PsSetLoadImageNotifyRoutine(PGImageNotify);
    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }
    g_ImageNotifyRegistered = TRUE;

    RtlZeroMemory(operations, sizeof(operations));
    operations[0].ObjectType = PsProcessType;
    operations[0].Operations = OB_OPERATION_HANDLE_CREATE | OB_OPERATION_HANDLE_DUPLICATE;
    operations[0].PreOperation = PGPreOperation;
    operations[0].PostOperation = PGPostOperation;

    operations[1].ObjectType = PsThreadType;
    operations[1].Operations = OB_OPERATION_HANDLE_CREATE | OB_OPERATION_HANDLE_DUPLICATE;
    operations[1].PreOperation = PGPreOperation;
    operations[1].PostOperation = PGPostOperation;

    RtlZeroMemory(&registration, sizeof(registration));
    registration.Version = OB_FLT_REGISTRATION_VERSION;
    registration.OperationRegistrationCount = 2;
    registration.Altitude = altitude;
    registration.OperationRegistration = operations;

    status = ObRegisterCallbacks(&registration, &g_ObRegistration);
    if (!NT_SUCCESS(status)) {
        PGUnload(DriverObject);
        return status;
    }
    g_ObRegistered = TRUE;

    return STATUS_SUCCESS;
}

#pragma pack(pop)

#define IOCTL_PG_SET_PID \
    CTL_CODE(FILE_DEVICE_UNKNOWN, 0x800, METHOD_BUFFERED, FILE_ANY_ACCESS)
#define IOCTL_PG_GET_ALERTS \
    CTL_CODE(FILE_DEVICE_UNKNOWN, 0x801, METHOD_BUFFERED, FILE_ANY_ACCESS)
#define IOCTL_PG_CLEAR_ALERTS \
    CTL_CODE(FILE_DEVICE_UNKNOWN, 0x802, METHOD_BUFFERED, FILE_ANY_ACCESS)
#define IOCTL_PG_GET_STATUS \
    CTL_CODE(FILE_DEVICE_UNKNOWN, 0x803, METHOD_BUFFERED, FILE_ANY_ACCESS)

static const char* AlertTypeName(ULONG type)
{
    switch (type) {
    case PG_ALERT_CONTROL: return "CONTROL";
    case PG_ALERT_PROCESS_CREATE: return "PROCESS_CREATE";
    case PG_ALERT_PROCESS_EXIT: return "PROCESS_EXIT";
    case PG_ALERT_IMAGE_LOAD: return "IMAGE_LOAD";
    case PG_ALERT_HANDLE_PRE: return "HANDLE_PRE";
    case PG_ALERT_HANDLE_POST: return "HANDLE_POST";
    default: return "UNKNOWN";
    }
}

static void PrintTime(ULONGLONG timestamp)
{
    FILETIME ft;
    SYSTEMTIME st;

    ft.dwLowDateTime = (DWORD)timestamp;
    ft.dwHighDateTime = (DWORD)(timestamp >> 32);

    if (FileTimeToSystemTime(&ft, &st)) {
        printf("%04u-%02u-%02u %02u:%02u:%02u",
            st.wYear, st.wMonth, st.wDay,
            st.wHour, st.wMinute, st.wSecond);
    }
    else {
        printf("%llu", timestamp);
    }
}

static void PrintAlert(const PG_ALERT& alert)
{
    printf("[");
    PrintTime(alert.Timestamp);
    printf("] %s", AlertTypeName(alert.Type));

    if (alert.ProcessId != 0) {
        printf(" pid=%lu", alert.ProcessId);
    }
    if (alert.TargetProcessId != 0) {
        printf(" target=%lu", alert.TargetProcessId);
    }
    if (alert.ProcessName[0] != L'\0') {
        printf(" process=%ls", alert.ProcessName);
    }
    if (alert.TargetName[0] != L'\0') {
        printf(" targetName=%ls", alert.TargetName);
    }
    if (alert.ImageName[0] != L'\0') {
        printf(" image=%ls", alert.ImageName);
    }
    if (alert.ImageBase != 0) {
        printf(" base=0x%llX size=0x%lX", alert.ImageBase, alert.ImageSize);
    }
    if (alert.Type == PG_ALERT_HANDLE_PRE) {
        printf(" access=0x%08lX -> 0x%08lX",
            alert.OriginalAccess, alert.DesiredAccess);
    }
    if (alert.Type == PG_ALERT_HANDLE_POST) {
        printf(" status=0x%08lX", (ULONG)alert.Status);
    }
    if (alert.Message[0] != L'\0') {
        printf(" | %ls", alert.Message);
    }
    printf("\n");
}

static HANDLE OpenDevice()
{
    HANDLE device = CreateFileW(
        PG_USER_PATH,
        GENERIC_READ | GENERIC_WRITE,
        0,
        nullptr,
        OPEN_EXISTING,
        FILE_ATTRIBUTE_NORMAL,
        nullptr);

    if (device == INVALID_HANDLE_VALUE) {
        printf("CreateFileW failed: %lu\n", GetLastError());
    }
    return device;
}

static bool SetProtectedPid(HANDLE device, ULONG pid)
{
    DWORD bytes = 0;

    if (!DeviceIoControl(device, IOCTL_PG_SET_PID,
        &pid, sizeof(pid), nullptr, 0, &bytes, nullptr)) {
        printf("IOCTL_PG_SET_PID failed: %lu\n", GetLastError());
        return false;
    }

    printf("Protected PID set to %lu\n", pid);
    return true;
}

static bool PrintStatus(HANDLE device)
{
    PG_STATUS status{};
    DWORD bytes = 0;

    if (!DeviceIoControl(device, IOCTL_PG_GET_STATUS,
        nullptr, 0, &status, sizeof(status), &bytes, nullptr)) {
        printf("IOCTL_PG_GET_STATUS failed: %lu\n", GetLastError());
        return false;
    }

    printf("Protected PID: %lu\n", status.ProtectedPid);
    printf("Pending alerts: %lu / %lu\n", status.AlertCount, status.MaxAlerts);
    return true;
}

static bool PrintAlerts(HANDLE device)
{
    PG_ALERT alerts[16]{};
    DWORD bytes = 0;

    if (!DeviceIoControl(device, IOCTL_PG_GET_ALERTS,
        nullptr, 0, alerts, sizeof(alerts), &bytes, nullptr)) {
        printf("IOCTL_PG_GET_ALERTS failed: %lu\n", GetLastError());
        return false;
    }

    const ULONG count = bytes / sizeof(PG_ALERT);
    for (ULONG i = 0; i < count; i++) {
        PrintAlert(alerts[i]);
    }
    return true;
}

int main(int argc, char** argv)
{
    if (argc < 2) {
        printf("Usage:\n");
        printf("  KernelCallBacksClient.exe set <pid>\n");
        printf("  KernelCallBacksClient.exe status\n");
        printf("  KernelCallBacksClient.exe alerts\n");
        printf("  KernelCallBacksClient.exe watch\n");
        printf("  KernelCallBacksClient.exe clear\n");
        return 1;
    }

    HANDLE device = OpenDevice();
    if (device == INVALID_HANDLE_VALUE) {
        return 1;
    }

    int result = 0;

    if (_stricmp(argv[1], "set") == 0) {
        if (argc < 3) {
            printf("Missing PID\n");
            result = 1;
        }
        else {
            ULONG pid = (ULONG)strtoul(argv[2], nullptr, 0);
            result = SetProtectedPid(device, pid) ? 0 : 1;
        }
    }
    else if (_stricmp(argv[1], "status") == 0) {
        result = PrintStatus(device) ? 0 : 1;
    }
    else if (_stricmp(argv[1], "alerts") == 0) {
        result = PrintAlerts(device) ? 0 : 1;
    }
    else if (_stricmp(argv[1], "clear") == 0) {
        DWORD bytes = 0;
        if (!DeviceIoControl(device, IOCTL_PG_CLEAR_ALERTS,
            nullptr, 0, nullptr, 0, &bytes, nullptr)) {
            printf("IOCTL_PG_CLEAR_ALERTS failed: %lu\n", GetLastError());
            result = 1;
        }
    }
    else if (_stricmp(argv[1], "watch") == 0) {
        printf("Watching alerts. Press Ctrl+C to stop.\n");
        for (;;) {
            PrintAlerts(device);
            Sleep(500);
        }
    }
    else {
        printf("Unknown command: %s\n", argv[1]);
        result = 1;
    }

    CloseHandle(device);
    return result;
}
